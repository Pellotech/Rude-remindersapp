javascriptfunction scoreResponse(message, rudenessLevel, taskText) {
  let score = 0;
  
  // Penalise blacklisted openers
  const weakOpeners = ['Time to', "It's time", "Don't forget", 
                       'Reminder:', 'Just a reminder'];
  if (weakOpeners.some(o => message.startsWith(o))) score -= 1;
  
  // Reward task name being used creatively
  const taskWords = taskText.toLowerCase().split(' ');
  if (taskWords.some(w => w.length > 3 && 
      message.toLowerCase().includes(w))) score += 2;
  
  // Reward questions (engaging)
  if (message.includes('?')) score += 1;
  
  // Reward level 4-5 having a punchline (comma after main clause)
  if (rudenessLevel >= 4 && message.includes(',')) score += 1;
  
  // Reward appropriate length — not too short, not too long
  const words = message.split(' ').length;
  if (words >= 8 && words <= 25) score += 1;
  
  return score;
}
