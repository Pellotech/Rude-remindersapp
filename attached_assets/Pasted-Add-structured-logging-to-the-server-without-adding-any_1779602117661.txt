A request logging middleware in server/index.ts that logs every incoming request with method, path, user ID if authenticated, and response time:

tsapp.use((req, res, next) => {
  const start = Date.now();
  res.on('finish', () => {
    log.info('http_request', {
      method: req.method,
      path: req.path,
      status: res.statusCode,
      latencyMs: Date.now() - start,
      userId: (req as any).user?.id,
    });
  });
  next();
});
